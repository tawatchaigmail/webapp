import { Component, signal } from '@angular/core';

@Component({
  selector: 'app-master-detail',
  template: `
    <div style="display: flex; gap: 20px;">
      <!-- Table 1: Departments (Master) -->
      <table border="1">
        <tr *ngFor="let dept of departments" (click)="selectDepartment(dept.id)" style="cursor:pointer">
          <td>{{ dept.name }}</td>
        </tr>
      </table>

      <!-- Table 2: Employees (Detail) -->
      <table border="1" *ngIf="filteredEmployees().length > 0">
        <tr *ngFor="let emp of filteredEmployees()">
          <td>{{ emp.name }}</td>
        </tr>
      </table>
    </div>
  `
})
export class MasterDetailComponent {
  departments = [{ id: 101, name: 'IT' }, { id: 102, name: 'HR' }];
  allEmployees = [
    { name: 'Alex', deptId: 101 },
    { name: 'Sarah', deptId: 101 },
    { name: 'John', deptId: 102 }
  ];

  // Using Angular Signals to store currently selected relationship ID
  selectedDeptId = signal<number | null>(null);

  selectDepartment(id: number) {
    this.selectedDeptId.set(id);
  }

  // Computed array automatically updates whenever selectedDeptId updates
  filteredEmployees = () => {
    const id = this.selectedDeptId();
    return id ? this.allEmployees.filter(e => e.deptId === id) : [];
  };
}
