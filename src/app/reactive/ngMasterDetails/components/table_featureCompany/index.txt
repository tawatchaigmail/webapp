import { Component, OnInit, ViewChild } from '@angular/core';
import { FacturaService } from '../../services/factura.service';
import { Factura } from '../../models/factura';
import { MatPaginator } from '@angular/material/paginator';
import { MatTableDataSource } from '@angular/material/table';
import { MatSort } from '@angular/material/sort';
import { Router } from '@angular/router';

@Component({
  selector: 'app-tabla-facturas',
  templateUrl: './tabla-facturas.component.html',
  styleUrls: ['./tabla-facturas.component.css']
})
export class TablaFacturasComponent implements OnInit {

  //arrayFacturas: Array<Factura>; //No lo borro por si quiero tener las facturas en un array
  nomCols: string[] = ['id', 'title'];
  facturas: MatTableDataSource<Factura>;
  cargado: boolean = false;
  idFacturaActual: string;
  idFacturaAnt: string;
  idFacturaSig: string;

  @ViewChild(MatPaginator, {static: true}) paginator: MatPaginator;
  @ViewChild(MatSort, {static: true}) sort: MatSort;                  

  constructor(private facturaService: FacturaService, private router: Router) {      
  }

  ngOnInit() {
    this.cargado = true;
    this.facturaService.recuperarFacturas().subscribe(
      facturas => {
        this.facturas = new MatTableDataSource<Factura>(facturas);
        this.facturas.paginator = this.paginator;
        this.facturas.sort = this.sort;
        this.cargado = true;
      },
      err => {
        console.log(err);
      }
    );
  }        

  aplicarFiltro(filterValue: string) {
    filterValue = filterValue.trim(); 
    filterValue = filterValue.toLowerCase();
    this.facturas.filter = filterValue;
  }

  verDetalle(posicion: number) {
    posicion = posicion + (this.facturas.paginator.pageIndex * this.facturas.paginator.pageSize);       
    this.router.navigate(['detalleFactura', posicion]);
  }

  recargarFacturas() {
    this.facturaService.recargarFacturas().subscribe(
      facturas => {
        this.facturas = new MatTableDataSource<Factura>(facturas);
        this.facturas.paginator = this.paginator;
        this.facturas.sort = this.sort;
        this.cargado = true;
      },
      err => {
        console.log(err);
      }
    );    
  }

  sortData(sort: MatSort) {
    console.log("sorting!");
  }
    
}
